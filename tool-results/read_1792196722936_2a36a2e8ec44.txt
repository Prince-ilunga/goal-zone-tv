     1→'use client'
     2→
     3→import { useEffect, useRef, useState } from 'react'
     4→import {
     5→  Dialog,
     6→  DialogContent,
     7→  DialogHeader,
     8→  DialogTitle,
     9→  DialogDescription,
    10→  DialogFooter,
    11→} from '@/components/ui/dialog'
    12→import { Button } from '@/components/ui/button'
    13→import { Input } from '@/components/ui/input'
    14→import { Label } from '@/components/ui/label'
    15→import { Textarea } from '@/components/ui/textarea'
    16→import {
    17→  Select,
    18→  SelectContent,
    19→  SelectItem,
    20→  SelectTrigger,
    21→  SelectValue,
    22→} from '@/components/ui/select'
    23→import { Checkbox } from '@/components/ui/checkbox'
    24→import { Tabs, TabsContent, TabsList, TabsTrigger } from '@/components/ui/tabs'
    25→import { Badge } from '@/components/ui/badge'
    26→import { toast } from 'sonner'
    27→import {
    28→  Lock,
    29→  LogOut,
    30→  Loader2,
    31→  Send,
    32→  Trash2,
    33→  Upload,
    34→  Image as ImageIcon,
    35→  Film,
    36→  Sparkles,
    37→  Link as LinkIcon,
    38→  Newspaper,
    39→  Download,
    40→  Save,
    41→} from 'lucide-react'
    42→import { SCOPES, SPORTS, SCOPE_META, SPORT_META, type Article, type Scope, type Sport } from '@/lib/types'
    43→import { slugify, formatDate, estimateReadMinutes } from '@/lib/format'
    44→
    45→interface AdminPanelProps {
    46→  open: boolean
    47→  onOpenChange: (open: boolean) => void
    48→}
    49→
    50→const TOKEN_KEY = 'gztv-token'
    51→const DRAFT_KEY = 'gztv-draft-v1'
    52→
    53→const DRAFT_DEFAULT = {
    54→  title: '',
    55→  excerpt: '',
    56→  content: '',
    57→  scope: 'NATIONAL' as Scope,
    58→  sport: 'FOOTBALL' as Sport,
    59→  location: '',
    60→  imageUrl: '',
    61→  videoUrl: '',
    62→  featured: false,
    63→  breaking: false,
    64→  source: '',
    65→}
    66→
    67→export function AdminPanel({ open, onOpenChange }: AdminPanelProps) {
    68→  const [token, setToken] = useState<string | null>(null)
    69→  const [password, setPassword] = useState('')
    70→  const [loginLoading, setLoginLoading] = useState(false)
    71→
    72→  const [draft, setDraft] = useState(DRAFT_DEFAULT)
    73→  const [saving, setSaving] = useState(false)
    74→  const [uploading, setUploading] = useState(false)
    75→  const [generating, setGenerating] = useState(false)
    76→  const [articles, setArticles] = useState<Article[]>([])
    77→  const [listLoading, setListLoading] = useState(false)
    78→
    79→  const fileInputRef = useRef<HTMLInputElement>(null)
    80→  const videoInputRef = useRef<HTMLInputElement>(null)
    81→
    82→  // Restore token + draft on mount
    83→  useEffect(() => {
    84→    if (typeof window === 'undefined') return
    85→    const t = localStorage.getItem(TOKEN_KEY)
    86→    if (t) setToken(t)
    87→    const d = localStorage.getItem(DRAFT_KEY)
    88→    if (d) {
    89→      try {
    90→        setDraft({ ...DRAFT_DEFAULT, ...JSON.parse(d) })
    91→      } catch {
    92→        // ignore
    93→      }
    94→    }
    95→  }, [])
    96→
    97→  // Persist draft whenever it changes
    98→  useEffect(() => {
    99→    if (typeof window === 'undefined') return
   100→    const t = setTimeout(() => {
   101→      localStorage.setItem(DRAFT_KEY, JSON.stringify(draft))
   102→    }, 400)
   103→    return () => clearTimeout(t)
   104→  }, [draft])
   105→
   106→  // Fetch articles when token becomes available
   107→  useEffect(() => {
   108→    if (token && open) fetchArticles()
   109→  }, [token, open])
   110→
   111→  const authHeaders = () => ({
   112→    'x-admin-token': token || '',
   113→  })
   114→
   115→  const onLogin = async (e: React.FormEvent) => {
   116→    e.preventDefault()
   117→    if (!password) {
   118→      toast.error('Veuillez saisir le mot de passe.')
   119→      return
   120→    }
   121→    setLoginLoading(true)
   122→    try {
   123→      const res = await fetch('/api/admin/login', {
   124→        method: 'POST',
   125→        headers: { 'Content-Type': 'application/json' },
   126→        body: JSON.stringify({ password }),
   127→      })
   128→      const data = await res.json()
   129→      if (!res.ok) throw new Error(data.error || 'Échec de connexion')
   130→      localStorage.setItem(TOKEN_KEY, data.token)
   131→      setToken(data.token)
   132→      setPassword('')
   133→      toast.success('Bienvenue, Moïse ! Accès rédaction accordé.')
   134→    } catch (err) {
   135→      toast.error(err instanceof Error ? err.message : 'Connexion impossible.')
   136→    } finally {
   137→      setLoginLoading(false)
   138→    }
   139→  }
   140→
   141→  const onLogout = () => {
   142→    localStorage.removeItem(TOKEN_KEY)
   143→    setToken(null)
   144→    setArticles([])
   145→    toast.info('Déconnexion réussie.')
   146→  }
   147→
   148→  const fetchArticles = async () => {
   149→    setListLoading(true)
   150→    try {
   151→      const res = await fetch('/api/admin-articles', {
   152→        headers: authHeaders(),
   153→        cache: 'no-store',
   154→      })
   155→      if (!res.ok) throw new Error()
   156→      const data = await res.json()
   157→      setArticles(data.articles || [])
   158→    } catch {
   159→      toast.error('Impossible de charger vos articles.')
   160→    } finally {
   161→      setListLoading(false)
   162→    }
   163→  }
   164→
   165→  const onUpload = async (
   166→    e: React.ChangeEvent<HTMLInputElement>,
   167→    type: 'image' | 'video',
   168→  ) => {
   169→    const file = e.target.files?.[0]
   170→    if (!file) return
   171→    setUploading(true)
   172→    try {
   173→      const fd = new FormData()
   174→      fd.append('file', file)
   175→      const res = await fetch('/api/upload', {
   176→        method: 'POST',
   177→        headers: authHeaders(),
   178→        body: fd,
   179→      })
   180→      const data = await res.json()
   181→      if (!res.ok) throw new Error(data.error || 'Upload échoué')
   182→      // L'API retourne mediaUrl (pas url) — support des 2 pour compatibilité
   183→      const mediaUrl = data.mediaUrl || data.url
   184→      if (!mediaUrl) throw new Error('URL du média manquante dans la réponse')
   185→      if (type === 'image') {
   186→        setDraft((d) => ({ ...d, imageUrl: mediaUrl }))
   187→        toast.success(`Image importée (${data.storage === 'cloudinary' ? 'Cloudinary' : 'local'}).`)
   188→      } else {
   189→        setDraft((d) => ({ ...d, videoUrl: mediaUrl }))
   190→        toast.success(`Vidéo importée (${data.storage === 'cloudinary' ? 'Cloudinary' : 'local'}).`)
   191→      }
   192→    } catch (err) {
   193→      toast.error(err instanceof Error ? err.message : 'Upload impossible.')
   194→    } finally {
   195→      setUploading(false)
   196→      if (fileInputRef.current) fileInputRef.current.value = ''
   197→      if (videoInputRef.current) videoInputRef.current.value = ''
   198→    }
   199→  }
   200→
   201→  // Téléchargement d'un média (image ou vidéo) vers la galerie / dossier Téléchargements
   202→  const downloadMedia = async (url: string, alt: string, isVideo: boolean) => {
   203→    if (!url) return
   204→    const baseName = (alt || 'goal-zone-tv').toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/^-+|-+$/g, '').slice(0, 40) || (isVideo ? 'video' : 'image')
   205→    const ext = isVideo ? 'mp4' : 'png'
   206→    const fileName = `${baseName}-${Date.now()}.${ext}`
   207→    toast.info(isVideo ? 'Téléchargement de la vidéo...' : "Téléchargement de l'image...")
   208→
   209→    // Méthode 1 : fetch + blob + <a download> (marche sur Chrome desktop + mobile récent)
   210→    try {
   211→      const fullUrl = url.startsWith('http') ? url : window.location.origin + url
   212→      const res = await fetch(fullUrl)
   213→      if (!res.ok) throw new Error('fetch failed')
   214→      const blob = await res.blob()
   215→      const objectUrl = URL.createObjectURL(blob)
   216→      const a = document.createElement('a')
   217→      a.href = objectUrl
   218→      a.download = fileName
   219→      a.style.display = 'none'
   220→      document.body.appendChild(a)
   221→      a.click()
   222→      document.body.removeChild(a)
   223→      setTimeout(() => URL.revokeObjectURL(objectUrl), 2000)
   224→      toast.success(
   225→        isVideo
   226→          ? 'Vidéo téléchargée. 📥 Vérifiez votre dossier Téléchargements / galerie.'
   227→          : 'Image téléchargée. 📥 Vérifiez votre dossier Téléchargements / galerie.'
   228→      )
   229→    } catch {
   230→      // Méthode 2 (fallback mobile) : ouvrir l'URL dans un nouvel onglet
   231→      // L'utilisateur fait un appui long > "Enregistrer l'image/vidéo"
   232→      const fullUrl = url.startsWith('http') ? url : window.location.origin + url
   233→      window.open(fullUrl, '_blank')
   234→      toast.info(
   235→        isVideo
   236→          ? 'Ouvrez la vidéo dans le nouvel onglet, appui long > Enregistrer la vidéo.'
   237→          : "Ouvrez l'image dans le nouvel onglet, appui long > Enregistrer l'image.",
   238→        { duration: 6000 }
   239→      )
   240→    }
   241→  }
   242→
   243→  const onGenerate = async () => {
   244→    if (!draft.title) {
   245→      toast.error('Saisissez un titre avant de générer une image IA.')
   246→      return
   247→    }
   248→    setGenerating(true)
   249→    try {
   250→      const res = await fetch('/api/generate-image', {
   251→        method: 'POST',
   252→        headers: { ...authHeaders(), 'Content-Type': 'application/json' },
   253→        body: JSON.stringify({
   254→          prompt: `Illustration journalistique pour un article de sport intitulé : "${draft.title}". Style photo éditorial, scène dynamique, haute qualité.`,
   255→        }),
   256→      })
   257→      const data = await res.json()
   258→      if (!res.ok) throw new Error(data.error || 'Génération échouée')
   259→      // L'API generate-image retourne imageUrl (pas url)
   260→      const mediaUrl = data.imageUrl || data.mediaUrl || data.url
   261→      if (!mediaUrl) throw new Error('URL image manquante')
   262→      setDraft((d) => ({ ...d, imageUrl: mediaUrl }))
   263→      toast.success('Image IA générée avec succès !')
   264→    } catch (err) {
   265→      toast.error(err instanceof Error ? err.message : 'Génération impossible.')
   266→    } finally {
   267→      setGenerating(false)
   268→    }
   269→  }
   270→
   271→  const onPublish = async (e: React.FormEvent) => {
   272→    e.preventDefault()
   273→    if (!draft.title || !draft.excerpt || !draft.content) {
   274→      toast.error('Titre, chapô et contenu sont obligatoires.')
   275→      return
   276→    }
   277→    setSaving(true)
   278→    try {
   279→      const res = await fetch('/api/articles', {
   280→        method: 'POST',
   281→        headers: { ...authHeaders(), 'Content-Type': 'application/json' },
   282→        body: JSON.stringify({
   283→          ...draft,
   284→          readMinutes: estimateReadMinutes(draft.content),
   285→        }),
   286→      })
   287→      const data = await res.json()
   288→      if (!res.ok) throw new Error(data.error || 'Publication échouée')
   289→      toast.success('Article publié avec succès !')
   290→      setDraft(DRAFT_DEFAULT)
   291→      localStorage.removeItem(DRAFT_KEY)
   292→      fetchArticles()
   293→    } catch (err) {
   294→      toast.error(err instanceof Error ? err.message : 'Publication impossible.')
   295→    } finally {
   296→      setSaving(false)
   297→    }
   298→  }
   299→
   300→  const onDelete = async (id: string) => {
   301→    if (!confirm('Supprimer cet article définitivement ?')) return
   302→    try {
   303→      const res = await fetch(`/api/articles?id=${id}`, {
   304→        method: 'DELETE',
   305→        headers: authHeaders(),
   306→      })
   307→      if (!res.ok) throw new Error()
   308→      setArticles(articles.filter((a) => a.id !== id))
   309→      toast.success('Article supprimé.')
   310→    } catch {
   311→      toast.error('Suppression impossible.')
   312→    }
   313→  }
   314→
   315→  const downloadUrl = (url: string | null | undefined) => {
   316→    if (!url) return
   317→    const a = document.createElement('a')
   318→    a.href = url
   319→    a.download = url.split('/').pop() || 'media'
   320→    a.click()
   321→  }
   322→
   323→  // Block dialog close while saving/uploading
   324→  const blockClose = (e: Event) => {
   325→    if (saving || uploading) e.preventDefault()
   326→  }
   327→
   328→  return (
   329→    <Dialog open={open} onOpenChange={onOpenChange}>
   330→      <DialogContent
   331→        showCloseButton={false}
   332→        className="max-h-[92vh] w-full max-w-3xl gap-0 overflow-hidden p-0 sm:max-w-3xl flex flex-col"
   333→        onPointerDownOutside={blockClose}
   334→        onInteractOutside={blockClose}
   335→        onEscapeKeyDown={blockClose}
   336→      >
   337→        <DialogHeader className="border-b bg-king-blue text-primary-foreground px-5 py-4 shrink-0">
   338→          <DialogTitle className="flex items-center gap-2 font-display text-xl">
   339→            <Lock className="size-5" /> Espace rédaction — GOAL ZONE TV
   340→          </DialogTitle>
   341→          <DialogDescription className="text-primary-foreground/80">
   342→            {token
   343→              ? 'Gérez vos articles : publier, lister, supprimer.'
   344→              : 'Accès réservé à Moïse Kalala. Saisissez votre mot de passe.'}
   345→          </DialogDescription>
   346→        </DialogHeader>
   347→
   348→        {!token ? (
   349→          /* ----- LOGIN ----- */
   350→          <form onSubmit={onLogin} className="space-y-4 p-6">
   351→            <div className="space-y-1.5">
   352→              <Label htmlFor="admin-pwd">Mot de passe</Label>
   353→              <Input
   354→                id="admin-pwd"
   355→                type="password"
   356→                value={password}
   357→                onChange={(e) => setPassword(e.target.value)}
   358→                placeholder="••••••••"
   359→                autoFocus
   360→              />
   361→            </div>
   362→            <Button
   363→              type="submit"
   364→              disabled={loginLoading}
   365→              className="w-full bg-king-blue"
   366→            >
   367→              {loginLoading ? (
   368→                <Loader2 className="mr-2 size-4 animate-spin" />
   369→              ) : (
   370→                <Lock className="mr-2 size-4" />
   371→              )}
   372→              Se connecter
   373→            </Button>
   374→            <p className="text-xs text-muted-foreground">
   375→              Astuce : raccourci clavier <kbd className="rounded bg-secondary px-1.5 py-0.5">Ctrl + Shift + E</kbd> pour ouvrir/fermer ce panneau.
   376→            </p>
   377→          </form>
   378→        ) : (
   379→          /* ----- TABS ----- */
   380→          <Tabs defaultValue="publish" className="flex w-full min-h-0 flex-1 flex-col">
   381→            <div className="flex shrink-0 items-center justify-between border-b px-3">
   382→              <TabsList className="bg-transparent">
   383→                <TabsTrigger value="publish">
   384→                  <Newspaper className="mr-1.5 size-4" /> Publier
   385→                </TabsTrigger>
   386→                <TabsTrigger value="list">
   387→                  Mes articles ({articles.length})
   388→                </TabsTrigger>
   389→              </TabsList>
   390→              <Button variant="ghost" size="sm" onClick={onLogout}>
   391→                <LogOut className="mr-1 size-3.5" /> Déconnexion
   392→              </Button>
   393→            </div>
   394→
   395→            {/* ----- PUBLISH TAB ----- */}
   396→            <TabsContent value="publish" className="m-0 min-h-0 flex-1 overflow-y-auto overscroll-contain" style={{ WebkitOverflowScrolling: 'touch' }}>
   397→              <form onSubmit={onPublish} className="space-y-4 p-5">
   398→                  <div className="space-y-1.5">
   399→                    <Label htmlFor="title">Titre *</Label>
   400→                    <Input
   401→                      id="title"
   402→                      value={draft.title}
   403→                      onChange={(e) => setDraft({ ...draft, title: e.target.value })}
   404→                      placeholder="Ex: TP Mazembe s'impose 3-1 face à Lupopo"
   405→                      maxLength={160}
   406→                    />
   407→                    {draft.title && (
   408→                      <p className="text-xs text-muted-foreground">
   409→                        Slug :{' '}
   410→                        <code className="rounded bg-secondary px-1 py-0.5">
   411→                          /articles/{slugify(draft.title)}
   412→                        </code>
   413→                      </p>
   414→                    )}
   415→                  </div>
   416→
   417→                  <div className="space-y-1.5">
   418→                    <Label htmlFor="excerpt">Chapô *</Label>
   419→                    <Textarea
   420→                      id="excerpt"
   421→                      value={draft.excerpt}
   422→                      onChange={(e) => setDraft({ ...draft, excerpt: e.target.value })}
   423→                      placeholder="Résumé court qui apparaîtra dans les cartes (2-3 phrases)."
   424→                      rows={3}
   425→                      maxLength={400}
   426→                    />
   427→                  </div>
   428→
   429→                  <div className="grid gap-4 sm:grid-cols-2">
   430→                    <div className="space-y-1.5">
   431→                      <Label htmlFor="scope">Portée</Label>
   432→                      <Select
   433→                        value={draft.scope}
   434→                        onValueChange={(v) => setDraft({ ...draft, scope: v as Scope })}
   435→                      >
   436→                        <SelectTrigger id="scope">
   437→                          <SelectValue />
   438→                        </SelectTrigger>
   439→                        <SelectContent>
   440→                          {SCOPES.map((s) => (
   441→                            <SelectItem key={s} value={s}>
   442→                              {SCOPE_META[s].label}
   443→                            </SelectItem>
   444→                          ))}
   445→                        </SelectContent>
   446→                      </Select>
   447→                    </div>
   448→                    <div className="space-y-1.5">
   449→                      <Label htmlFor="sport">Discipline</Label>
   450→                      <Select
   451→                        value={draft.sport}
   452→                        onValueChange={(v) => setDraft({ ...draft, sport: v as Sport })}
   453→                      >
   454→                        <SelectTrigger id="sport">
   455→                          <SelectValue />
   456→                        </SelectTrigger>
   457→                        <SelectContent>
   458→                          {SPORTS.map((s) => (
   459→                            <SelectItem key={s} value={s}>
   460→                              {SPORT_META[s].emoji} {SPORT_META[s].label}
   461→                            </SelectItem>
   462→                          ))}
   463→                        </SelectContent>
   464→                      </Select>
   465→                    </div>
   466→                  </div>
   467→
   468→                  <div className="grid gap-4 sm:grid-cols-2">
   469→                    <div className="space-y-1.5">
   470→                      <Label htmlFor="location">Lieu</Label>
   471→                      <Input
   472→                        id="location"
   473→                        value={draft.location}
   474→                        onChange={(e) => setDraft({ ...draft, location: e.target.value })}
   475→                        placeholder="Lubumbashi, Kinshasa, Le Caire..."
   476→                        maxLength={80}
   477→                      />
   478→                    </div>
   479→                    <div className="space-y-1.5">
   480→                      <Label htmlFor="source">Source (optionnel)</Label>
   481→                      <Input
   482→                        id="source"
   483→                        value={draft.source}
   484→                        onChange={(e) => setDraft({ ...draft, source: e.target.value })}
   485→                        placeholder="AFP, CAF, LINAFOOT..."
   486→                        maxLength={80}
   487→                      />
   488→                    </div>
   489→                  </div>
   490→
   491→                  <div className="space-y-1.5">
   492→                    <Label htmlFor="content">Contenu *</Label>
   493→                    <Textarea
   494→                      id="content"
   495→                      value={draft.content}
   496→                      onChange={(e) => setDraft({ ...draft, content: e.target.value })}
   497→                      placeholder="Rédigez votre article ici. Séparez les paragraphes par des sauts de ligne."
   498→                      rows={8}
   499→                      maxLength={15000}
   500→                    />
   501→                    <p className="text-xs text-muted-foreground">
   502→                      ≈ {estimateReadMinutes(draft.content)} min de lecture ·{' '}
   503→                      {draft.content.split(/\s+/).filter(Boolean).length} mots
   504→                    </p>
   505→                  </div>
   506→
   507→                  {/* Médias */}
   508→                  <div className="space-y-2 rounded-md border p-3">
   509→                    <div className="flex items-center gap-2">
   510→                      <ImageIcon className="size-4 text-primary" />
   511→                      <span className="text-sm font-semibold">Image à la une</span>
   512→                    </div>
   513→                    <div className="flex flex-wrap items-center gap-2">
   514→                      <input
   515→                        ref={fileInputRef}
   516→                        type="file"
   517→                        accept="image/*"
   518→                        onChange={(e) => onUpload(e, 'image')}
   519→                        className="hidden"
   520→                      />
   521→                      <Button
   522→                        type="button"
   523→                        variant="outline"
   524→                        size="sm"
   525→                        onClick={() => fileInputRef.current?.click()}
   526→                        disabled={uploading}
   527→                      >
   528→                        {uploading ? (
   529→                          <Loader2 className="mr-1 size-3.5 animate-spin" />
   530→                        ) : (
   531→                          <Upload className="mr-1 size-3.5" />
   532→                        )}
   533→                        Importer image
   534→                      </Button>
   535→                      <Button
   536→                        type="button"
   537→                        variant="outline"
   538→                        size="sm"
   539→                        onClick={onGenerate}
   540→                        disabled={generating}
   541→                      >
   542→                        {generating ? (
   543→                          <Loader2 className="mr-1 size-3.5 animate-spin" />
   544→                        ) : (
   545→                          <Sparkles className="mr-1 size-3.5" />
   546→                        )}
   547→                        Générer IA
   548→                      </Button>
   549→                      <div className="relative flex-1 min-w-[200px]">
   550→                        <LinkIcon className="pointer-events-none absolute left-3 top-1/2 size-3.5 -translate-y-1/2 text-muted-foreground" />
   551→                        <Input
   552→                          value={draft.imageUrl}
   553→                          onChange={(e) => setDraft({ ...draft, imageUrl: e.target.value })}
   554→                          placeholder="https://... (URL image)"
   555→                          className="h-9 pl-8 text-xs"
   556→                        />
   557→                      </div>
   558→                    </div>
   559→                    {draft.imageUrl && (
   560→                      <div className="relative mt-2 aspect-video w-full overflow-hidden rounded-md bg-muted">
   561→                        <img
   562→                          src={draft.imageUrl}
